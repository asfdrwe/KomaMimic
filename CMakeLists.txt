cmake_minimum_required(VERSION 3.22)
if(DEFINED KOMAMIMIC_EXTENDED_DOCUMENTS)
    message(STATUS "KOMAMIMIC_EXTENDED_DOCUMENTS is obsolete in v0.5; Qt WebEngine is always included")
    unset(KOMAMIMIC_EXTENDED_DOCUMENTS CACHE)
endif()
# Migrate the old vcpkg feature; WebEngine is now a required dependency.
list(REMOVE_ITEM VCPKG_MANIFEST_FEATURES extended-documents)
include(cmake/WindowsFast.cmake)
project(KomaMimic VERSION 1.0.3 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
add_library(komamimic_options INTERFACE)
option(KOMAMIMIC_APP_SYMBOLS "Keep symbols for KomaMimic's own code in Release" ${KOMAMIMIC_WINDOWS_FAST})
target_compile_definitions(komamimic_options INTERFACE KOMAMIMIC_VERSION="${PROJECT_VERSION}")
if(MSVC)
    # Both source literals and the execution character set must be UTF-8.
    target_compile_options(komamimic_options INTERFACE /W4 /utf-8 /permissive- /Zc:__cplusplus)
    if(KOMAMIMIC_APP_SYMBOLS)
        target_compile_options(komamimic_options INTERFACE /Zi)
        target_link_options(komamimic_options INTERFACE /DEBUG /OPT:REF /OPT:ICF)
    endif()
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(komamimic_options INTERFACE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    target_compile_definitions(komamimic_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
include(cmake/Homebrew.cmake)
if(KOMAMIMIC_WINDOWS_FAST)
    find_package(Qt6 ${KOMAMIMIC_QT_VERSION} EXACT REQUIRED COMPONENTS Widgets Concurrent Network PrintSupport Sql Qml WebEngineWidgets)
else()
    find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Concurrent Network PrintSupport Sql Qml WebEngineWidgets)
endif()
find_package(Threads REQUIRED)
find_package(LibArchive REQUIRED)
find_package(ICU REQUIRED COMPONENTS uc)
find_package(ZLIB REQUIRED)
include(cmake/Poppler.cmake)
find_package(SDL2 2.0.14 REQUIRED CONFIG)
if(TARGET SDL2::SDL2)
    set(KOMAMIMIC_SDL_TARGET SDL2::SDL2)
elseif(TARGET SDL2::SDL2-static)
    set(KOMAMIMIC_SDL_TARGET SDL2::SDL2-static)
else()
    message(FATAL_ERROR "SDL2 package must provide SDL2::SDL2 or SDL2::SDL2-static")
endif()
if(UNIX AND NOT APPLE)
    find_package(Qt6 QUIET COMPONENTS DBus)
endif()
add_library(reader_core src/i18n.cpp src/book.cpp src/reading.cpp src/filenames.cpp src/archivenames.cpp src/media.cpp src/imageprocessing.cpp src/documentsource.cpp src/publication.cpp src/pdfsource.cpp src/apng.cpp src/expression.cpp src/zipmetadata.cpp src/scriptreader.cpp src/readers.qrc)
include(cmake/Translations.cmake)
target_sources(reader_core PRIVATE src/archivepath.cpp)
target_include_directories(reader_core PUBLIC src)
target_link_libraries(reader_core PUBLIC komamimic_options Qt6::Gui LibArchive::LibArchive ICU::uc ZLIB::ZLIB komamimic_pdf_dependencies)
add_library(reader_ui src/pageview.cpp src/shortcuts.cpp src/mainwindow.cpp src/namesdialog.cpp src/thumbnailgrid.cpp src/foldertree.cpp src/processingdialog.cpp src/mainwindownavigation.cpp src/mainwindowimage.cpp src/librarystore.cpp src/bookexport.cpp src/readerfeatures.cpp src/readerfeatureio.cpp src/readerfeatureprefs.cpp src/featurecommands.cpp src/readerinput.cpp src/startup.cpp src/power.cpp src/idleinhibitor.cpp src/inhibitprotocol.cpp src/collectiondetails.cpp src/documentsettings.cpp src/mainwindowdocument.cpp)
target_sources(reader_ui PRIVATE src/defaultpreferences.cpp src/legacysettings.cpp src/legacysettingsdialog.cpp src/readertoolbar.cpp src/mainwindowmenus.cpp src/thumbnailcache.cpp src/thumbnailsource.cpp src/appearance.cpp)
target_link_libraries(reader_ui PUBLIC reader_core Qt6::Widgets Qt6::Concurrent Qt6::Network Qt6::PrintSupport Qt6::Sql ${KOMAMIMIC_SDL_TARGET})
# Qt owns main() and event handling; SDL is used for input/audio only.
target_compile_definitions(reader_ui PUBLIC SDL_MAIN_HANDLED)
if(WIN32)
    target_sources(reader_ui PRIVATE src/power_windows.cpp)
    target_link_libraries(reader_ui PRIVATE kernel32)
elseif(APPLE)
    find_library(KOMAMIMIC_IOKIT_FRAMEWORK IOKit REQUIRED)
    find_library(KOMAMIMIC_COREFOUNDATION_FRAMEWORK CoreFoundation REQUIRED)
    target_sources(reader_ui PRIVATE src/power_macos.cpp)
    target_link_libraries(reader_ui PRIVATE ${KOMAMIMIC_IOKIT_FRAMEWORK} ${KOMAMIMIC_COREFOUNDATION_FRAMEWORK})
else()
    target_sources(reader_ui PRIVATE src/power_linux.cpp)
endif()
if(TARGET Qt6::DBus AND UNIX AND NOT APPLE)
 target_link_libraries(reader_ui PRIVATE Qt6::DBus)
 target_compile_definitions(reader_ui PRIVATE KOMAMIMIC_DBUS=1)
endif()
target_sources(reader_ui PRIVATE src/documentview.cpp src/documents.qrc)
target_link_libraries(reader_ui PUBLIC Qt6::WebEngineWidgets)
add_executable(komamimic src/main.cpp)
target_link_libraries(komamimic PRIVATE reader_ui)
if(WIN32)
    set_target_properties(komamimic PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
include(cmake/ApplicationIcons.cmake)
add_executable(komamimic-script-worker src/scriptworker.cpp)
target_link_libraries(komamimic-script-worker PRIVATE reader_core Qt6::Qml Qt6::Gui Threads::Threads)
add_dependencies(komamimic komamimic-script-worker)
if(BUILD_TESTING)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Test)
    add_subdirectory(tests)
endif()
install(TARGETS komamimic komamimic-script-worker RUNTIME DESTINATION bin)
install(DIRECTORY scripts/readers DESTINATION share/KomaMimic)

install(FILES LICENSE README.md
    DESTINATION share/doc/KomaMimic)
